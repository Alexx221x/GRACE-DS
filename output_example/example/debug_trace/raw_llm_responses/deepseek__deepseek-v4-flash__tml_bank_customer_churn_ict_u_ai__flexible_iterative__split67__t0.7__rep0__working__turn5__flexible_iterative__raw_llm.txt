from sklearn.pipeline import Pipeline
from sklearn.compose import ColumnTransformer
from sklearn.preprocessing import OneHotEncoder, StandardScaler, FunctionTransformer
from sklearn.impute import SimpleImputer
from sklearn.ensemble import RandomForestClassifier
import pandas as pd
import numpy as np

def engineer_features(X):
    X = X.copy()
    X['AgeGroup'] = pd.cut(X['Age'], bins=[0, 30, 45, 60, 100], labels=[0,1,2,3]).astype(int)
    X['BalanceSalaryRatio'] = X['Balance'] / (X['EstimatedSalary'] + 1e-5)
    X['TenureAgeRatio'] = X['Tenure'] / (X['Age'] + 1e-5)
    X['ProdsXActive'] = X['NumOfProducts'] * X['IsActiveMember']
    # Drop identifier columns not useful for modelling
    drop_cols = ['CustomerId', 'Surname']
    X = X.drop(columns=[c for c in drop_cols if c in X.columns])
    return X

numeric_features = ['CreditScore', 'Age', 'Tenure', 'Balance', 'NumOfProducts',
                    'HasCrCard', 'IsActiveMember', 'EstimatedSalary',
                    'AgeGroup', 'BalanceSalaryRatio', 'TenureAgeRatio', 'ProdsXActive']
categorical_features = ['Geography', 'Gender']

pipeline = Pipeline(steps=[
    ('engineer', FunctionTransformer(engineer_features, validate=False)),
    ('preprocessor', ColumnTransformer(transformers=[
        ('num', Pipeline(steps=[
            ('imputer', SimpleImputer(strategy='median')),
            ('scaler', StandardScaler())
        ]), numeric_features),
        ('cat', Pipeline(steps=[
            ('imputer', SimpleImputer(strategy='most_frequent')),
            ('onehot', OneHotEncoder(handle_unknown='ignore', sparse_output=False))
        ]), categorical_features)
    ])),
    ('classifier', RandomForestClassifier(n_estimators=200, max_depth=10,
                                          class_weight='balanced', random_state=42))
])
